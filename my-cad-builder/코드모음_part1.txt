SYSTEM_PROMPT = """너는 블록 기반 건축물 구조 시뮬레이션 결과를 분석하는 구조 안전 분석가야.

- 출력은 한국어 평문(마크다운 기호 없이). 문제점 3~5개를 번호로 쓰고,
  각 항목마다 '근거 수치 / 위치(높이·구역) / 개선 제안'을 한 줄씩 써.
- 마지막에 '가장 먼저 손볼 곳' 한 줄로 마무리해."""


def parse_block_id(block_id: str):
    """BlockID = X_Z_Y (각각 부호문자 + 숫자, 값*10). 예: -405_0615_0015 -> (-40.5, 1.5, 61.5)"""
    try:
        seg_x, seg_z, seg_y = block_id.split("_")
        vals = []
        for seg in (seg_x, seg_y, seg_z):
            sign = -1 if seg[0] == "-" else 1
            vals.append(sign * int(seg[1:]) / 10)
        return tuple(vals)  # (x, y, z)
    except Exception:
        return (float("nan"),) * 3


def load_blocks(csv_bytes: bytes) -> tuple[pd.DataFrame, int]:
    """파괴 블록이 DESTROYED 행으로 중복 기록되는 CSV를 BlockID당 1행으로 합친다."""
    try:
        df = pd.read_csv(io.BytesIO(csv_bytes))
    except Exception as e:
        raise HTTPException(status_code=400, detail=f"CSV를 읽을 수 없습니다: {e}")

    missing = REQUIRED_COLS - set(df.columns)
    if missing:
        raise HTTPException(
            status_code=400,
            detail=f"필수 컬럼이 없습니다: {', '.join(sorted(missing))}",
        )

    df["_destroyed"] = df["RiskLevel"].eq("Destroyed")
    for col in ("PosX", "PosY", "PosZ", "Stress"):
        df[col] = pd.to_numeric(df[col], errors="coerce")  # 'DESTROYED' -> NaN

    # 좌표가 살아있는 첫 행이 원본 정보(Material/Tool 등)를 가진다
    base = (
        df.dropna(subset=["PosX", "PosY", "PosZ"])
        .drop_duplicates("BlockID", keep="first")
        .set_index("BlockID")
    )
    agg = df.groupby("BlockID").agg(
        max_stress=("Stress", "max"),
        destroyed=("_destroyed", "any"),
    )
    blocks = base.drop(columns=["_destroyed", "Stress"]).join(agg, how="left")
    blocks["final_risk"] = blocks["RiskLevel"].where(~blocks["destroyed"], "Destroyed")

    blocks = blocks.reset_index()
    orig = blocks["BlockID"].map(parse_block_id)
    blocks["orig_x"] = orig.map(lambda t: t[0])
    blocks["orig_y"] = orig.map(lambda t: t[1])
    blocks["orig_z"] = orig.map(lambda t: t[2])
    # 현재 좌표(PosX/Y/Z)는 테스트 후 위치라서, ID의 원래 좌표와의 거리 = 밀려난 정도
    blocks["moved"] = (
        (blocks["PosX"] - blocks["orig_x"]) ** 2
        + (blocks["PosY"] - blocks["orig_y"]) ** 2
        + (blocks["PosZ"] - blocks["orig_z"]) ** 2
    ) ** 0.5

    orphans = df["BlockID"].nunique() - len(blocks)  # 좌표 행이 없는 ID
    return blocks, int(orphans)


def _risk_counts(frame: pd.DataFrame) -> dict:
    return {k: int(v) for k, v in frame["final_risk"].value_counts().items()}


def build_summary(blocks: pd.DataFrame, orphans: int) -> dict:
    total = len(blocks)

    by_tool = {}
    for tool, g in blocks.groupby("Tool"):
        by_tool[tool] = {
            "count": int(len(g)),
            "risk": _risk_counts(g),
            "destroyed_pct": round(float(g["destroyed"].mean()) * 100, 1),
            "median_stress": round(float(g["max_stress"].median()), 2),
        }

    by_height = []
    bands = pd.cut(blocks["orig_y"], bins=5)
    for band, g in blocks.groupby(bands, observed=True):
        by_height.append({
            "y_range": f"{band.left:.1f}~{band.right:.1f}",
            "count": int(len(g)),
            "destroyed_pct": round(float(g["destroyed"].mean()) * 100, 1),
            "median_stress": round(float(g["max_stress"].median()), 2),
        })

    zones = blocks.assign(
        zx=pd.cut(blocks["orig_x"], bins=3, labels=["X낮음", "X중간", "X높음"]),
        zz=pd.cut(blocks["orig_z"], bins=3, labels=["Z낮음", "Z중간", "Z높음"]),
    )
    zone_stats = (
        zones.groupby(["zx", "zz"], observed=True)["destroyed"]
        .agg(["sum", "count"])
        .sort_values("sum", ascending=False)
        .head(3)
    )
    top_zones = [
        {"zone": f"{zx}/{zz}", "destroyed": int(r["sum"]), "blocks": int(r["count"])}
        for (zx, zz), r in zone_stats.iterrows()
    ]

    shifted = blocks[blocks["moved"] > MOVED_THRESHOLD]
    movement = {
        "threshold": MOVED_THRESHOLD,
        "moved_count": int(len(shifted)),
        "moved_pct": round(len(shifted) / total * 100, 1),
        "moved_median": round(float(shifted["moved"].median()), 1) if len(shifted) else 0.0,
        "moved_max": round(float(blocks["moved"].max()), 1),
        "moved_by_risk": _risk_counts(shifted),
    }

    cols = ["BlockID", "orig_x", "orig_y", "orig_z", "moved", "max_stress", "Tool", "Material", "final_risk"]
    hotspots = (
        blocks.sort_values("max_stress", ascending=False)
        .head(10)[cols]
        .round(2)
        .to_dict("records")
    )

    reinf = blocks[blocks["Tool"] == "Reinforcement"]
    failed_reinf = reinf[reinf["final_risk"].isin(["Danger", "Destroyed"])]

    return {
        "total_blocks": int(total),
        "orphan_destroyed_ids": orphans,
        "risk": _risk_counts(blocks),
        "stress_median": round(float(blocks["max_stress"].median()), 2),
        "stress_max": round(float(blocks["max_stress"].max()), 2),
        "by_tool": by_tool,
        "by_height": by_height,
        "top_destroyed_zones": top_zones,
        "reinforcement_failed_count": int(len(failed_reinf)),
        "reinforcement_total": int(len(reinf)),
        "movement": movement,
        "hotspots": hotspots,
    }


def model_chain() -> list[str]:
    custom = os.getenv("GEMINI_MODEL")
    return ([custom] if custom else []) + [m for m in DEFAULT_MODELS if m != custom]


def _post(headers: dict, model: str, summary: dict, low_effort: bool):
    body = {
        "model": model,
        "messages": [
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": str(summary)},
        ],
        "temperature": 0.3,
        "max_tokens": 4096,
    }
    if low_effort:
        body["reasoning_effort"] = "low"  # 사고 시간을 줄여서 타임아웃 방지
    return requests.post(LLM_URL, headers=headers, json=body, timeout=LLM_TIMEOUT)


def call_llm(summary: dict) -> tuple[str, str]:
    key = (os.getenv("GEMINI_API_KEY") or os.getenv("GOOGLE_API_KEY") or "").strip().strip("\"'")
    if not key:
        raise RuntimeError("GEMINI_API_KEY가 .env에 설정되지 않았습니다.")

    headers = {"Authorization": f"Bearer {key}", "Content-Type": "application/json"}
    errors = []
    for model in model_chain():
        try:
            resp = _post(headers, model, summary, low_effort=True)
            if resp.status_code == 400 and "reasoning" in resp.text.lower():
                resp = _post(headers, model, summary, low_effort=False)  # 이 모델은 옵션 미지원
        except requests.exceptions.Timeout:
            errors.append(f"{model}: 시간초과")
            continue

        if resp.status_code in (400, 401, 403):
            raise RuntimeError(
                f"{resp.status_code} 인증/요청 오류 - 서버가 읽은 키: {key[:4]}...{key[-3:]} (길이 {len(key)}). "
                f"Gemini 키는 보통 'AIza'로 시작하고 39자입니다. 응답: {resp.text[:200]}"
            )
        if resp.status_code == 429:
            raise RuntimeError("429 무료 한도 초과입니다. 잠시 후(1분 뒤) 다시 시도하세요.")
        if resp.status_code in (404, 500, 503):  # 없는 모델/일시 장애 -> 다음 모델
            errors.append(f"{model}: {resp.status_code}")
            continue
        resp.raise_for_status()
        message = resp.json()["choices"][0]["message"]
        text = THINK_RE.sub("", message.get("content") or "").strip()
        if text:
            return text, model
        errors.append(f"{model}: 빈 응답")

    raise RuntimeError("사용 가능한 모델이 없습니다 -> " + ", ".join(errors))


@router.post("/api/analyze-stress")
async def analyze_stress(file: UploadFile = File(...)):
    csv_bytes = await file.read()
    blocks, orphans = load_blocks(csv_bytes)
    summary = build_summary(blocks, orphans)

    analysis, analysis_error, model_used = None, None, None
    try:
        analysis, model_used = await run_in_threadpool(call_llm, summary)
    except Exception as e:
        # LLM이 실패해도 통계는 돌려준다
        analysis_error = str(e)

    return {
        "stats": summary,
        "analysis": analysis,
        "analysis_error": analysis_error,
        "model": model_used,
    }

MAX_BLOCKS = 10000
GRID_PITCH = 3.0
UNITS_PER_METER = 20.0
APPLY_TILT_FIX = True   # False로 바꾸면 기울기 보정 끄기 (A/B 비교용)


def format_id(val):
    abs_val = round(abs(val))
    sign = "-" if val < 0 else "0"
    return f"{sign}{str(abs_val).zfill(3)}"


def get_mesh_from_colab(img_bytes: bytes) -> trimesh.Trimesh:

def straighten_tilt(mesh, up_axis_idx=1, mode="shear", low=0.2, high=0.8,
                    bins=24, min_deg=0.5, max_deg=30.0):
    """
    건물 중간 구간(높이 20~80%)의 층별 중심점을 이어 '쏠린 정도(기울기)'를 구하고,
    그만큼 되돌려 건물을 똑바로 세운다.
      mode="shear"  : 각 높이의 층을 옆으로 밀어 중심선을 수직으로 (층은 수평 유지) ← 블록 변환용 기본값
      mode="rotate" : 메쉬 전체를 회전 (건물 전체가 통째로 기운 경우)
    기울기가 너무 작거나(min_deg) 너무 크면(max_deg, 추정 실패로 판단) 원본 유지.
    """
    verts = np.asarray(mesh.vertices, dtype=np.float64).copy()
    up = up_axis_idx
    hor = [i for i in range(3) if i != up]
    h = verts[:, up]
    h_min, h_max = h.min(), h.max()
    H = h_max - h_min
    if H < 1e-6:
        return mesh

    # 1. 높이 구간별 중심점
    edges = np.linspace(h_min + low * H, h_min + high * H, bins + 1)
    ys, cs = [], []
    for a, b in zip(edges[:-1], edges[1:]):
        m = (h >= a) & (h < b)
        if m.sum() < 20:
            continue
        ys.append(h[m].mean())
        cs.append(verts[m][:, hor].mean(axis=0))
    if len(ys) < 5:
        print("📏 기울기 보정 건너뜀: 중심점이 부족함")
        return mesh
    ys, cs = np.array(ys), np.array(cs)

    # 2. 두 수평축 각각 직선 피팅 (튀는 점 1회 제거)
    slopes = []
    for j in range(2):
        k, b0 = np.polyfit(ys, cs[:, j], 1)
        res = cs[:, j] - (k * ys + b0)
        keep = np.abs(res) <= max(2 * res.std(), 1e-9)
        if keep.sum() >= 4:
            k, b0 = np.polyfit(ys[keep], cs[keep, j], 1)
        slopes.append(k)
    slopes = np.array(slopes)
    tilt = float(np.degrees(np.arctan(np.linalg.norm(slopes))))

    if tilt < min_deg or tilt > max_deg:
        print(f"📏 기울기 보정 건너뜀: 기울기 {tilt:.2f}° (허용 {min_deg}~{max_deg}° 밖)")
        return mesh

    # 3. 되돌리기
    if mode == "rotate":
        u = np.zeros(3)
        u[up] = 1.0
        u[hor] = slopes
        u /= np.linalg.norm(u)
        t = np.zeros(3)
        t[up] = 1.0
        v = np.cross(u, t)
        c = float(np.dot(u, t))
        s = float(np.linalg.norm(v))
        vx = np.array([[0, -v[2], v[1]], [v[2], 0, -v[0]], [-v[1], v[0], 0]])
        R = np.eye(3) + vx + vx @ vx * ((1 - c) / (s ** 2))
        pivot = verts[h <= h_min + 0.02 * H].mean(axis=0)   # 바닥 중심 기준
        verts = (verts - pivot) @ R.T + pivot
    else:
        verts[:, hor] -= np.outer(h - h_min, slopes)         # 바닥은 고정, 위로 갈수록 되밀기

    mesh.vertices = verts
    print(f"📏 기울기 보정: {tilt:.2f}° 되돌림 ({mode})")
    return mesh


def mesh_to_blocks(mesh: trimesh.Trimesh, real_height_m: float, max_blocks: int = MAX_BLOCKS):
    extents = mesh.bounds[1] - mesh.bounds[0]
    tallest_axis = int(np.argmax(extents))

    if tallest_axis == 2:
        rot = trimesh.transformations.rotation_matrix(
            angle=np.radians(-90), direction=[1, 0, 0], point=mesh.centroid
        )
        mesh.apply_transform(rot)
    elif tallest_axis == 0:
        rot = trimesh.transformations.rotation_matrix(
            angle=np.radians(90), direction=[0, 0, 1], point=mesh.centroid
        )
        mesh.apply_transform(rot)

    mirror = trimesh.transformations.scale_matrix(-1, direction=[0, 0, 1], origin=mesh.centroid)
    mesh.apply_transform(mirror)
    mesh.invert()

    print(f"📏 축 보정: extents={extents}, tallest_axis={tallest_axis}")

    # 🌟 기울기(쏠림) 보정: 여기서는 축 정리가 끝나 y가 높이인 상태라 축을 추측할 필요가 없다.
    #    (균등 스케일이라 스케일링 전/후 어디서 해도 결과 동일)
    if APPLY_TILT_FIX:
        mesh = straighten_tilt(mesh, up_axis_idx=1, mode="shear")

    mesh_height_raw = mesh.bounds[1][1] - mesh.bounds[0][1]
    if mesh_height_raw <= 0:
        raise ValueError("메시 높이를 계산할 수 없습니다 (생성된 메시가 비정상일 수 있음)")

    target_height_units = real_height_m * UNITS_PER_METER
    mesh.apply_scale(target_height_units / mesh_height_raw)
    mesh.apply_translation([0, -mesh.bounds[0][1], 0])

    surface_area = mesh.area
    pitch = max(GRID_PITCH, round(((surface_area / max_blocks) ** 0.5) / GRID_PITCH) * GRID_PITCH)

    raw_blocks = []
    for attempt in range(5):
        estimated_cells = surface_area / (pitch ** 2)
        sample_count = max(int(estimated_cells * 6), 5000)
        points, face_indices = trimesh.sample.sample_surface(mesh, count=sample_count)
        normals = mesh.face_normals[face_indices]

        raw_blocks = []
        seen_ids = set()
        for (x, y, z), normal in zip(points, normals):
            grid_x = round((x - 1.5) / pitch)
            grid_y = round((y - 1.5) / pitch)
            grid_z = round((z - 1.5) / pitch)

            pos_x = grid_x * GRID_PITCH + 1.5
            pos_y = grid_y * GRID_PITCH + 1.5
            pos_z = grid_z * GRID_PITCH + 1.5

            id_x = format_id(pos_x * 10)
            id_z = format_id(pos_z * 10)
            id_y = format_id(pos_y * 10)
            block_id = f"{id_x}_{id_z}_{id_y}"
            if block_id in seen_ids:
                continue
            seen_ids.add(block_id)

            block_type = "Floor" if abs(normal[1]) > 0.7 else "Wall"

            raw_blocks.append({
                "BlockID": block_id,
                "PosX": round(pos_x, 2),
                "PosY": round(pos_y, 2),
                "PosZ": round(pos_z, 2),
                "Stress": 0.00,
                "RiskLevel": "Safe",
                "Prescription": "N",
                "Material": "Default",
                "Tensile": 400.0,
                "Compressive": 400.0,
                "Tool": "Existing",
                "Type": block_type,
            })

        print(f"📐 시도 {attempt + 1}: pitch={pitch} → {len(raw_blocks)}개 블록")
        if len(raw_blocks) <= max_blocks:
            break
        pitch += GRID_PITCH

    return raw_blocks

# 🌟 지붕 모서리 A/B/C/D 자동 추정 (카메라와의 수평 거리 기준 정렬)

def order_roof_corners_by_distance(mesh, x_min, x_max, z_min, z_max):
    """
    지붕 4개 모서리를 카메라와 가까운 순서대로 A→D 로 라벨링.
    - 거리 판정은 X,Z 평면(수평)만 사용. Y(높이)는 결과값이라 거리 계산에서 반드시 제외.
    - main.py의 미러링 보정 이후 좌표계 기준, Z가 큰 쪽이 카메라(정면)에 가깝다고 가정.
    """
    verts = mesh.vertices
    y_max = verts[:, 1].max()
    y_min = verts[:, 1].min()
    roof_mask = verts[:, 1] >= y_max - (y_max - y_min) * 0.10
    roof_verts = verts[roof_mask]

    if len(roof_verts) == 0:
        roof_verts = verts  # 안전장치: 지붕 영역이 안 잡히면 전체 정점으로 대체

    corners_xz = {
        "TL": (x_min, z_min),
        "TR": (x_max, z_min),
        "BL": (x_min, z_max),
        "BR": (x_max, z_max),
    }

    corner_info = {}
    for key, (cx, cz) in corners_xz.items():
        # 1. 이 모서리 위치(cx, cz)에 가장 가까운 실제 지붕 꼭짓점 찾기 (X,Z만 비교)
        dists_to_corner = np.hypot(roof_verts[:, 0] - cx, roof_verts[:, 2] - cz)
        nearest = roof_verts[dists_to_corner.argmin()]
        corner_info[key] = {
            "pos": (float(nearest[0]), float(nearest[1]), float(nearest[2])),
        }

    # 2. 카메라(정면, Z가 큰 방향)에 가까운 순서로 정렬 — Z값 큰 것부터
    ordered_keys = sorted(corner_info.keys(), key=lambda k: -corner_info[k]["pos"][2])

    label_map = {}
    for i, key in enumerate(ordered_keys):
        letter = chr(ord("A") + i)  # A, B, C, D
        label_map[letter] = {
            "original_position": key,
            "estimated_height_m": round(corner_info[key]["pos"][1] / UNITS_PER_METER, 2),
        }
    return label_map


# ====================================================================
# 🌟 지붕 높이 보정: 평지붕(기본값) 또는 4개 모서리 보간(상세입력)
# ====================================================================
def _make_fill_block(pos_x: float, pos_y: float, pos_z: float, block_type: str, seen_ids: set):
    """roof 보정으로 뜬 틈을 메울 Wall 블록 하나 생성. 이미 그 자리에 블록이 있으면 None."""
    id_x = format_id(pos_x * 10)
    id_z = format_id(pos_z * 10)
    id_y = format_id(pos_y * 10)
    block_id = f"{id_x}_{id_z}_{id_y}"
    if block_id in seen_ids:
        return None
    seen_ids.add(block_id)
    return {
        "BlockID": block_id,
        "PosX": round(pos_x, 2),
        "PosY": round(pos_y, 2),
        "PosZ": round(pos_z, 2),
        "Stress": 0.00,
        "RiskLevel": "Safe",
        "Prescription": "N",
        "Material": "Default",
        "Tensile": 400.0,
        "Compressive": 400.0,
        "Tool": "Existing",
        "Type": block_type,
    }


def apply_roof_correction(raw_blocks: list, corner_heights: dict | None = None):
    """
    corner_heights가 없으면(아무 입력 없음): 모서리 4개 높이가 같다고 가정 → 평지붕으로 강제.
    corner_heights가 있으면: {"tl":.., "tr":.., "bl":.., "br":..} (미터) 로 bilinear interpolation.

    🌟 Floor 블록을 새 높이로 옮기기만 하면 옛 높이와 새 높이 사이가 뚫려서
       "몸체와 분리된 지붕(뚜껑)"이 생긴다. 그 틈을 Wall 블록으로 채워서 이어준다.
    """
    if not raw_blocks:
        return raw_blocks

    xs = [b["PosX"] for b in raw_blocks]
    zs = [b["PosZ"] for b in raw_blocks]
    ys = [b["PosY"] for b in raw_blocks]
    x_min, x_max = min(xs), max(xs)
    z_min, z_max = min(zs), max(zs)
    y_max, y_min = max(ys), min(ys)

    roof_threshold = y_max - (y_max - y_min) * 0.10
    roof_blocks = [b for b in raw_blocks if b["Type"] == "Floor" and b["PosY"] >= roof_threshold]
    roof_ids = {id(b) for b in roof_blocks}
    other_blocks = [b for b in raw_blocks if id(b) not in roof_ids]

    if not roof_blocks:
        return raw_blocks

    seen_ids = {b["BlockID"] for b in raw_blocks}

    if corner_heights is None:
        def target_height(x, z):
            return round(y_max, 2)
        print(f"🏠 지붕 상세입력 없음 → 모서리 4개 동일 높이({round(y_max, 2)})로 평지붕 보정")
    else:
        tl = corner_heights["tl"] * UNITS_PER_METER
        tr = corner_heights["tr"] * UNITS_PER_METER
        bl = corner_heights["bl"] * UNITS_PER_METER
        br = corner_heights["br"] * UNITS_PER_METER

        def target_height(x, z):
            u = 0.0 if x_max == x_min else (x - x_min) / (x_max - x_min)
            v = 0.0 if z_max == z_min else (z - z_min) / (z_max - z_min)
            top = tl + (tr - tl) * u
            bot = bl + (br - bl) * u
            return round(top + (bot - top) * v, 2)
        print(f"🏠 모서리 4개 입력값으로 지붕 보간 완료 (tl={tl}, tr={tr}, bl={bl}, br={br})")

    fill_blocks = []
    for b in roof_blocks:
        old_y = b["PosY"]
        new_y = target_height(b["PosX"], b["PosZ"])

        if new_y > old_y:
            # 옛 지붕 높이 → 새 지붕 높이 사이를 Wall로 채워서 몸체와 연결
            y = old_y + GRID_PITCH
            while y < new_y - 1e-6:
                fb = _make_fill_block(b["PosX"], y, b["PosZ"], "Wall", seen_ids)
                if fb:
                    fill_blocks.append(fb)
                y += GRID_PITCH
        elif new_y < old_y:
            # 지붕을 낮추는 경우: 원래 위치를 Wall로 남겨서 단면이 뚫리지 않게 함
            fb = _make_fill_block(b["PosX"], old_y, b["PosZ"], "Wall", seen_ids)
            if fb:
                fill_blocks.append(fb)

        b["PosY"] = new_y

    return other_blocks + roof_blocks + fill_blocks

@app.post("/api/estimate-roof-corners")
async def estimate_roof_corners(
    file: UploadFile = File(...),
    startX: float = Form(...),
    startY: float = Form(...),
    endX: float = Form(...),
    endY: float = Form(...),
):
    img_bytes = await file.read()
    try:
        image = Image.open(io.BytesIO(img_bytes)).convert("RGB")
    except Exception as e:
        raise HTTPException(status_code=400, detail=f"이미지를 읽을 수 없습니다: {e}")

    sx, sy = max(0, int(startX)), max(0, int(startY))
    ex, ey = min(image.width, int(endX)), min(image.height, int(endY))
    cropped = image.crop((sx, sy, ex, ey))

    crop_bytes_io = io.BytesIO()
    cropped.save(crop_bytes_io, format="PNG")
    crop_bytes = crop_bytes_io.getvalue()

    mesh = get_mesh_from_colab(crop_bytes)

    x_min, x_max = float(mesh.bounds[0][0]), float(mesh.bounds[1][0])
    z_min, z_max = float(mesh.bounds[0][2]), float(mesh.bounds[1][2])

    label_map = order_roof_corners_by_distance(mesh, x_min, x_max, z_min, z_max)
    return label_map
    # 예: {"A": {"original_position": "BR", "estimated_height_m": 12.3}, "B": {...}, "C": {...}, "D": {...}}


@app.post("/api/extract-3d-csv")
async def extract_3d_csv(
    file: UploadFile = File(...),
    startX: float = Form(...),
    startY: float = Form(...),
    endX: float = Form(...),
    endY: float = Form(...),
    realHeightMeters: float = Form(...),
    # 🌟 지붕 모서리 4개 높이(미터) — 상세입력 안 하면 전부 None → 평지붕 기본값
    roofTL: float | None = Form(None),
    roofTR: float | None = Form(None),
    roofBL: float | None = Form(None),
    roofBR: float | None = Form(None),
):
    print(f"📥 사진 데이터 및 박스 좌표 수신: ({startX}, {startY}) ~ ({endX}, {endY}) | 실제 높이: {realHeightMeters}m")

    img_bytes = await file.read()
    try:
        image = Image.open(io.BytesIO(img_bytes)).convert("RGB")
    except Exception as e:
        raise HTTPException(
            status_code=400,
            detail=f"이미지를 읽을 수 없습니다 (지원하지 않는 포맷일 수 있음): {e}",
        )

    sx, sy = max(0, int(startX)), max(0, int(startY))
    ex, ey = min(image.width, int(endX)), min(image.height, int(endY))
    cropped = image.crop((sx, sy, ex, ey))

    crop_bytes_io = io.BytesIO()
    cropped.save(crop_bytes_io, format="PNG")
    crop_bytes = crop_bytes_io.getvalue()

    print("🤖 콜랍 AI 모델에 메시 생성 요청 중...")
    mesh = get_mesh_from_colab(crop_bytes)
    print(f"✅ 메시 수신 완료 (vertices={len(mesh.vertices)}, faces={len(mesh.faces)})")

    raw_blocks = mesh_to_blocks(mesh, real_height_m=realHeightMeters)

    # 🌟 지붕 보정: 4개 값이 모두 들어오면 상세 보간, 하나라도 비어있으면 평지붕 기본값
    corner_heights = None
    if None not in (roofTL, roofTR, roofBL, roofBR):
        corner_heights = {"tl": roofTL, "tr": roofTR, "bl": roofBL, "br": roofBR}
    raw_blocks = apply_roof_correction(raw_blocks, corner_heights)

    df = pd.DataFrame(raw_blocks)
    df = df.drop_duplicates(subset=["BlockID"])
    df = df.sort_values(by=["PosZ", "PosX", "PosY"])

    print(f"✅ 총 {len(df)}개의 3D 빌딩 블록 엑셀 변환 완료! (속은 비어있는 표면 전용 구조)")

    output = io.StringIO()
    df.to_csv(output, index=False)
    output.seek(0)

    return StreamingResponse(
        io.BytesIO(output.getvalue().encode("utf-8")),
        media_type="text/csv",
        headers={"Content-Disposition": "attachment; filename=ai_extracted_3d.csv"},
    )
